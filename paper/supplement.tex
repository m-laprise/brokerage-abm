\section*{Supplementary figures}

\begin{figure}[htbp]
\centering
\includegraphics[width=\linewidth]{type_geometry.png}
\caption{\textbf{Principal types and their latent curve.} All \pv{suppDgpN} baseline types from realization \pv{suppDgpHeatmapSeed}, projected onto pairs of the curve's first \pv{settingDgpProjectionWord} principal components (A--C). Gray points are realized types; the line shows the latent curve, colored by position along it. Matching and learning use the full \pv{settingDgpDimensionsWord}-dimensional types, not these projections.}
\label{fig:supp-type-geometry}
\end{figure}

\begin{figure}[htbp]
\centering
\includegraphics[width=\linewidth]{match_value_surfaces.png}
\caption{\textbf{Underlying match value across pairs of principal types.} \pv{suppDgpHeatmapN} types selected at evenly spaced ranks of individual quality from realization \pv{suppDgpHeatmapSeed} (\pv{suppDgpN} principals). Types and matching-function objects $c$, $A$, and $B$ are fixed across panels. Individual quality increases rightward and upward. Upper triangles exclude self-matches; colors show each pair's underlying value minus the grand mean of the full value matrix.}
\label{fig:supp-dgp-structure}
\end{figure}

\begin{figure}[htbp]
\centering
\includegraphics[width=\linewidth]{effective_dimensionality.png}
\caption{\textbf{Dimensionality of underlying match value.} Matrices of pair-specific underlying match values, centered within each realization and summarized over \pv{suppDgpSeeds} independent realizations. Types and matching-function draws are fixed within each realization as $\rho$ and $\delta$ vary. A: the first \pv{suppDgpSpectrumComponents} singular values at $\delta=\pv{suppDgpDelta}$, divided by the largest within each realization. B: the fewest components accounting for \pv{suppDgpRankEnergyPercent}\% of the sum of squared singular values. Lines and markers are means; ribbons and whiskers show pointwise \pv{settingSuppIntervalPercent}\% Monte Carlo intervals. N/A denotes $\rho=1$, where underlying match value depends only on individual quality and difficulty is inapplicable.}
\label{fig:supp-dgp-dimension}
\end{figure}

\begin{figure}[htbp]
\centering
\includegraphics[width=\linewidth]{complementarity_contributions.png}
\caption{\textbf{Contributions of assessment and access across matching composition.} Late-mean differences paired by seed: A, full service minus access-only; B, full service minus assessment-only. Parameters other than matching composition and turnover remain at baseline. Points average \pv{aaOtherSeeds} seeds per regime; bars show pointwise \pv{aaIntervalPercent}\% Monte Carlo intervals. Connecting lines are visual guides.}
\label{fig:supp-assessment-access}
\end{figure}

\begin{figure}[htbp]
\centering
\includegraphics[width=\linewidth]{alternative_measures_composition_turnover.png}
\caption{\textbf{Structural position and access under alternative measures.} Ensemble means from the neural-network simulations, using constraint and effective size at period \pv{settingLateNetworkSnapshot} and late-window averages for access and degree. Parameters other than matching composition and turnover remain at baseline. Lines connect matching compositions at fixed turnover, with $\rho=1$ shown separately. Principal degree includes the broker tie when present. Whiskers show pointwise \pv{settingSuppIntervalPercent}\% Monte Carlo intervals.}
\label{fig:supp-structure}
\end{figure}

\begin{figure}[htbp]
\centering
\includegraphics[width=\linewidth]{alternative_measures_experiments.png}
\caption{\textbf{Structural consequences of services and learning under alternative measures.} A--B: differences paired by seed across matching composition and turnover, with other parameters at baseline. Adding assessment is full service minus access-only; adding access is full service minus assessment-only. Dashed lines connect contrasts within each regime; the inset enlarges the region near zero. C--D: ensemble means from the baseline ridge experiments. Constraint and effective size use period \pv{settingLateNetworkSnapshot}; other measures use late-window averages. Whiskers show pointwise \pv{settingSuppIntervalPercent}\% Monte Carlo intervals.}
\label{fig:supp-experiments}
\end{figure}

\begin{figure}[htbp]
\centering
\includegraphics[width=\linewidth]{learning_across_difficulties.png}
\caption{\textbf{Sources of the broker's advantage across matching difficulties.} Ensemble late means from the ridge experiments; parameters other than $\rho$ and $\delta$ remain at baseline. Ranking differences and net-output shares are calculated within seed before averaging. Whiskers show pointwise \pv{settingSuppIntervalPercent}\% Monte Carlo intervals. The same $\rho=1$ estimates appear in each column because difficulty has no effect there. Lines are visual guides.}
\label{fig:supp-learning}
\end{figure}
