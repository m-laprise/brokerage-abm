The mutable simulation state at the end of period $t$ is
\[
S^t=\Bigl(
X^t,G^t,\gamma,
\{ \mathcal H_i^t,\Theta_i^t,\bar q_{i\cdot}^t,C_{i\cdot}^t,
s_i^t,\chi_i^t,M_i^t\}_{i=1}^N,
\mathcal H_b^t,\Theta_b^t,\mathrm{Roster}^t,D^t,
\mathrm{rep}_b^t
\Bigr).
\]

The matching function is
\[
f(x_i,x_j)
=\rho\,\frac{x_i^\top c+x_j^\top c}{2}
+(1-\rho)\,g(x_i,x_j)\,x_i^\top A x_j,
\qquad
g(x_i,x_j)=1+\delta\,\operatorname{sign}(x_i^\top Bx_j).
\]
\section{Top-Level Simulation}

\begin{breakablealgorithm}
\caption{Brokerage Simulation}
\label{alg:simulation}
\begin{algorithmic}[1]
\Require Parameters $\vartheta$ and seed.
\Ensure Final state $S^T$ and \diag{measure table $\mathcal M\in\R^{T\times m}$}.
\State $S^0\gets\Call{\procInitialize}{\vartheta,\mathrm{seed}}$
\For{$t=1,\ldots,T$}
    \State $(S^t,\mathcal R^t)\gets\Call{\procPeriodUpdate}{S^{t-1},t,\vartheta}$
    \State \diag{$\mathcal M_{t\cdot}\gets\Call{\procCollectMeasures}{S^t,\mathcal R^t,\vartheta}$}
\EndFor
\State \Return $(S^T,\diag{\mathcal M})$
\end{algorithmic}
\end{breakablealgorithm}

\section{Training}

For any one-hidden-layer predictor with input dimension $p_{\mathrm{in}}$ and hidden width $h$,
\[
\mathrm{NN}_{\Theta}(\xi)=w_2^\top\relu(W_1\xi+b_1)+b_2,\qquad
\Theta=(W_1,b_1,w_2,b_2),
\]
where $\xi\in\R^{p_{\mathrm{in}}}$, $W_1\in\R^{h\times p_{\mathrm{in}}}$,
$b_1,w_2\in\R^h$, and $b_2\in\R$.
\[
d_b=d+\frac{d(d+1)}{2},\qquad
\psi_b(x_i,x_j)=
\left[
x_i+x_j;\;
\operatorname{vech}\!\left(\frac{x_ix_j^\top+x_jx_i^\top}{2}\right)
\right].
\]
Agent networks use $(p_{\mathrm{in}},h)=(d,2d)$; the broker network uses
$(p_{\mathrm{in}},h)=(d_b,8d)$.

\[
\psi_b(x_i,x_j)=\psi_b(x_j,x_i)
\]
so broker histories produce one training column per observed unordered pair.

\begin{localvartable}{training}
$p_{\mathrm{in}},h$ & integers & Predictor input dimension and hidden width. \\
$\xi$ & $\R^{p_{\mathrm{in}}}$ & Predictor input vector. \\
$\Theta$ & NN parameters & Generic MLP parameters $(W_1,b_1,w_2,b_2)$. \\
$W_1,b_1,w_2,b_2$ & matrix, vectors, scalar & One-hidden-layer MLP weights and biases. \\
$d_b$ & integer & Broker symmetric pair-feature dimension. \\
$\psi_b$ & function & Symmetric broker pair-feature map. \\
$\Xi_{\mathrm{tr}},y_{\mathrm{tr}}$ & $\R^{p_{\mathrm{in}}\times n}$, $\R^n$ & Generic training inputs and targets. \\
$\mathcal L$ & scalar & Full-batch MSE objective. \\
$e$ & integer & Update-step index. \\
$(m,v,\tau)$ & Adam state & Persistent Adam moments and step counter; $A_i,A_b$ are the agent/broker instances. \\
$\beta_1,\beta_2,\epsilon$ & scalars & Adam constants $(0.9,\,0.999,\,10^{-8})$. \\
$n_i,n_b$ & integers & Current values of history lengths $n_i^t$ and $n_b^t$. \\
$L_i,L_b$ & ordered index sets & Training-window indices: observations from the most recent $W_p$ periods, subsampled evenly to at most $M$. \\
$\Delta_i,\Delta_b$ & integers & New observations since last training call. \\
$E_i,E_b$ & integers & Agent and broker update-step counts. \\
$\Xi_{i,\mathrm{tr}},y_{i,\mathrm{tr}}$ & $\R^{d\times |L_i|}$, $\R^{|L_i|}$ & Agent $i$ training view of $\mathcal H_i^t$. \\
$\Psi_{b,\mathrm{tr}},y_{b,\mathrm{tr}}$ & $\R^{d_b\times |L_b|}$, $\R^{|L_b|}$ & Broker symmetric-feature training view of $\mathcal H_b^t$. \\
\end{localvartable}

\begin{breakablealgorithm}
\caption{\procTrainNN}
\label{alg:trainnn}
\begin{algorithmic}[1]
\Require Network parameters $\Theta$, inputs $\Xi_{\mathrm{tr}}\in\R^{p_{\mathrm{in}}\times n}$, targets $y_{\mathrm{tr}}\in\R^n$,
steps $E$, learning rate $\eta_{\mathrm{lr}}$, persistent Adam state $(m,v,\tau)$.
\For{$e=1,\ldots,E$}
    \State $g\gets\nabla_{\Theta}\mathcal L(\Theta)$, \quad
    $\mathcal L(\Theta)= n^{-1}\sum_{\ell=1}^n
    \left(\mathrm{NN}_{\Theta}(\Xi_{\mathrm{tr},\cdot\ell})-y_{\mathrm{tr},\ell}\right)^2$
    \State $\tau\gets\tau+1$
    \State $m\gets\beta_1 m+(1-\beta_1)g$, \quad $v\gets\beta_2 v+(1-\beta_2)\,g\odot g$
    \State $\Theta\gets\Theta-\eta_{\mathrm{lr}}\,
    \dfrac{m/(1-\beta_1^{\tau})}{\sqrt{v/(1-\beta_2^{\tau})}+\epsilon}$
\EndFor
\State \Return $\Theta,(m,v,\tau)$
\end{algorithmic}
\end{breakablealgorithm}

Within period-level routines, state variables without a superscript refer to
their current-period values.

\begin{breakablealgorithm}
\caption{\procTrainPredictors}
\label{alg:trainpredictors}
\begin{algorithmic}[1]
\Require State $S$, period $t$, parameters $(E_{\mathrm{init}},E_{\mathrm{steps}},\eta_{\mathrm{lr}},W_p,M)$.
\For{$i=1,\ldots,N$ in parallel}
    \State $n_i\gets n_i^t$, $\Delta_i\gets$ observations added to $\mathcal H_i$ since its last training call.
    \If{$n_i>0$, $\Delta_i>0$, and $i\bmod 2=t\bmod 2$}
        \State $L_i\gets$ indices of $\mathcal H_i$ observations from the most recent $W_p$ periods; if $|L_i|>M$, keep $M$ spaced evenly across $L_i$.
        \State $\Xi_{i,\mathrm{tr}}\gets(\Xi_i^t)_{\cdot L_i}$, $y_{i,\mathrm{tr}}\gets(y_i^t)_{L_i}$.
        \State $E_i\gets\max\{E_{\mathrm{steps}},\lceil E_{\mathrm{init}}\Delta_i/n_i\rceil\}$.
        \State $(\Theta_i,A_i)\gets\Call{\procTrainNN}{\Theta_i,\Xi_{i,\mathrm{tr}},y_{i,\mathrm{tr}},E_i,\eta_{\mathrm{lr}},A_i}$; $\Delta_i\gets0$.
    \EndIf
\EndFor
\State $n_b\gets n_b^t$, $\Delta_b\gets$ broker observations since last training.
\If{$n_b>0$ and $\Delta_b>0$}
    \State $L_b\gets$ indices of $\mathcal H_b$ observations from the most recent $W_p$ periods; if $|L_b|>M$, keep $M$ spaced evenly across $L_b$.
    \State $\Psi_{b,\mathrm{tr}}\gets\left[\psi_b\!\left((X_{b,1}^t)_{\cdot\ell},(X_{b,2}^t)_{\cdot\ell}\right)\right]_{\ell\in L_b}$.
    \State $y_{b,\mathrm{tr}}\gets(y_b^t)_{L_b}$.
    \State $E_b\gets\max\{E_{\mathrm{steps}},\lceil E_{\mathrm{init}}\Delta_b/n_b\rceil\}$.
    \State $(\Theta_b,A_b)\gets\Call{\procTrainNN}{\Theta_b,\Psi_{b,\mathrm{tr}},y_{b,\mathrm{tr}},E_b,\eta_{\mathrm{lr}},A_b}$; $\Delta_b\gets0$.
\EndIf
\end{algorithmic}
\end{breakablealgorithm}

\section{Initialization}

Calibration draws random pairs $(i_\ell,j_\ell)$ and sets
\[
\begin{aligned}
\bar q_{\mathrm{cal}}
&=\frac{1}{10{,}000}\sum_{\ell=1}^{10{,}000} \{Q+f(x_{i_\ell},x_{j_\ell})\},\\
r&=f_r\,\bar q_{\mathrm{cal}},
\qquad
\phi=c_s=\lambda_c\,\bar q_{\mathrm{cal}}.
\end{aligned}
\]

\begin{localvartable}{\procInitialize}
$\lambda_c$ & scalar & Cost rate used to set $\phi$ and $c_s$. \\
$f_r$ & scalar & Reservation coefficient; sets $r=f_r\,\bar q_{\mathrm{cal}}$ (default 0.60). \\
$(i_\ell,j_\ell)$ & pair in $[N]^2$ & Random calibration pair. \\
$d_\gamma$ & integer & Number of active curve dimensions. \\
$p_{\mathrm{rewire}}$ & scalar & Watts-Strogatz rewiring probability. \\
$\nu_\ell$ & integer & Frequency for active curve coordinate $\ell$. \\
$\psi_\ell$ & scalar & Phase for active curve coordinate $\ell$. \\
$A_0$ & $\R^{d\times d}$ & Raw matrix used to construct $A$. \\
$\Sigma_x$ & $\R^{d\times d}$ & Empirical agent type second moment. \\
$B_0$ & $\R^{d\times d}$ & Raw zero-trace symmetric regime matrix. \\
$B_{\mathrm{raw}}$ & $\R^{d\times d}$ & Orthogonalized pre-normalization regime matrix. \\
$\Psi_{b,\mathrm{seed}},y_{b,\mathrm{seed}}$ & $\R^{d_b\times n_b^0}$, $\R^{n_b^0}$ & Broker seed training batch using symmetric pair features. \\
\end{localvartable}

\begin{localvartable}{\procInitializeAgent}
$\sigma_x$ & scalar & Agent type noise scale. \\
$u_i$ & scalar & Agent $i$ curve position. \\
$\epsilon_i$ & $\R^d$ & Agent type noise for agent $i$. \\
$\pi_{ij}^{\mathrm{ent}}$ & scalar & Entrant attachment probability. \\
$\mathcal V_i^{\mathrm{ent}}$ & subset of $[N]\setminus\{i\}$ & Entrant attachment set. \\
\end{localvartable}

\begin{breakablealgorithm}
\caption{\procInitializeAgent}
\label{alg:initializeagent}
\begin{algorithmic}[1]
\Require State $S$, agent slot $i$, period counter $t$, parameters $\vartheta$.
\Ensure Agent slot $i$ is initialized in $S$.
\State Draw $u_i\sim\mathrm{Unif}[0,1]$ and
$\epsilon_i\sim \mathcal N(0,\sigma_x^2 I_d/d)$.
\State $x_i^t\gets(\gamma(u_i)+\epsilon_i)/\|\gamma(u_i)+\epsilon_i\|_2$.
\State Reset $\mathcal H_i^t=(\Xi_i^t,y_i^t,n_i^t)$ to zero observations, $\bar q_{i\cdot}^t$, $C_{i\cdot}^t$,
$\chi_i^t$, and $M_i^t$; draw fresh agent MLP weights $\Theta_i^t$ with output bias $Q$.
\If{$t=0$}
    \State $s_i^t\gets(0,0)$.
\Else
    \State Remove $i$ from $\mathrm{Roster}^t$, $D^t$, and $M_j^t$ for all $j$;
    delete all incident edges in $G^t$; set $\bar q_{ji}^t,C_{ji}^t\gets0$ for all $j\ne i$.
    \State Sample $\mathcal V_i^{\mathrm{ent}}\subset[N]\setminus\{i\}$ without replacement,
    $|\mathcal V_i^{\mathrm{ent}}|=\min(\lfloor k_G/2\rfloor,N-1)$, with
    \[
    \pi_{ij}^{\mathrm{ent}}
    \propto \exp\{-\|x_i^t-x_j^t\|_2^2\}.
    \]
    \State Add edges $(i,j)$ to $G^t$ for all $j\in\mathcal V_i^{\mathrm{ent}}$.
    \State $s_{i,\mathrm{self}}^t\gets
    \begin{cases}
    |\mathcal V_i^{\mathrm{ent}}|^{-1}\sum_{j\in\mathcal V_i^{\mathrm{ent}}}s_{j,\mathrm{self}}^t,
    & \mathcal V_i^{\mathrm{ent}}\ne\emptyset,\\
    0, & \mathcal V_i^{\mathrm{ent}}=\emptyset.
    \end{cases}$
    \State $s_{i,\mathrm{broker}}^t\gets\mathrm{rep}_b^t$.
\EndIf
\end{algorithmic}
\end{breakablealgorithm}

\begin{breakablealgorithm}
\caption{\procInitialize}
\label{alg:initialize}
\begin{algorithmic}[1]
\Require Parameters $\vartheta$ and seed.
\Ensure Initial state $S^0$.
\State Draw curve frequencies $\nu_\ell\sim\mathrm{Unif}\{1,\ldots,5\}$ and phases
$\psi_\ell\sim\mathrm{Unif}[0,2\pi)$ for $\ell=1,\ldots,d_\gamma$, defining $\gamma$.
\State Allocate empty agent and broker containers in $S^0$ and store $\gamma$.
\For{$i=1,\ldots,N$}
    \State $\Call{\procInitializeAgent}{S^0,i,0,\vartheta}$.
\EndFor
\State Draw ideal agent type $c$ as a noisy curve point.
\State Draw $A_0\in\R^{d\times d}$ iid standard normal and set
$A\gets A_0^\top A_0\cdot d/\operatorname{tr}(A_0^\top A_0)$.
\State $\Sigma_x\gets N^{-1}\sum_i x_i^0(x_i^0)^\top$.
\State Draw symmetric zero-trace $B_0$ and set
\[
B_{\mathrm{raw}}\gets B_0-
\frac{\operatorname{tr}(\Sigma_xB_0\Sigma_xA)}{\operatorname{tr}(\Sigma_xA\Sigma_xA)}A,\qquad
B\gets B_{\mathrm{raw}}/\|B_{\mathrm{raw}}\|_F .
\]
\State Calibrate $(\bar q_{\mathrm{cal}},r,\phi,c_s)$.
\State Build $G^0$ as a Watts-Strogatz graph on $[N]$ with degree $k_G$ and
rewiring probability $p_{\mathrm{rewire}}$; add isolated broker node $N+1$.
\State Initialize $\mathcal H_b^0$, $\Theta_b^0$, and $\mathrm{rep}_b^0$.
\State Sample $\mathrm{Roster}^0\subset[N]$ uniformly with
$|\mathrm{Roster}^0|=\lceil\alpha_R N\rceil$; add broker edges $(i,N+1)$ for all roster members.
\For{each unordered non-broker edge $\{i,j\}\in\EG^0$}
    \State Draw $q_{ij}$ once; append $(x_j^0,q_{ij})$ to $\mathcal H_i^0$ and
    $(x_i^0,q_{ij})$ to $\mathcal H_j^0$; update
    $\bar q_{ij}^0,C_{ij}^0,\bar q_{ji}^0,C_{ji}^0$.
\EndFor
\State Let $\mathcal B_{\mathrm{seed}}\gets
\{\{i,j\}\in\EG^0:i,j\in\mathrm{Roster}^0\}$; shuffle and keep at most 100 edges.
\For{each $\{i,j\}\in\mathcal B_{\mathrm{seed}}$}
    \State Append $(x_i^0,x_j^0,q_{ij})$ to $\mathcal H_b^0$; increment $n_b^0$.
\EndFor
\If{$n_b^0>0$}
    \State Set $\mathrm{rep}_b^0$ to mean broker seed output.
\Else
    \State Set $\mathrm{rep}_b^0\gets0$.
\EndIf
\State Set $s_{i,\mathrm{self}}^0$ to mean own seed output and
$s_{i,\mathrm{broker}}^0\gets\mathrm{rep}_b^0$.
\For{each $i$ with $n_i^0>0$}
    \State $\Theta_i^0\gets\Call{\procTrainNN}{\Theta_i^0,\Xi_i^0,y_i^0,E_{\mathrm{init}},\eta_{\mathrm{lr}}}$.
\EndFor
\If{$n_b^0>0$}
    \State $\Psi_{b,\mathrm{seed}}\gets\left[\psi_b\!\left((X_{b,1}^0)_{\cdot\ell},(X_{b,2}^0)_{\cdot\ell}\right)\right]_{\ell=1}^{n_b^0}$,
    $y_{b,\mathrm{seed}}\gets y_b^0$.
    \State $\Theta_b^0\gets\Call{\procTrainNN}{\Theta_b^0,\Psi_{b,\mathrm{seed}},y_{b,\mathrm{seed}},E_{\mathrm{init}},\eta_{\mathrm{lr}}}$.
\EndIf
\State \Return $S^0$.
\end{algorithmic}
\end{breakablealgorithm}

\section{Channel Choice}

\begin{localvartable}{\procChooseChannel}
$U_{\mathrm{self}}$ & scalar & Self-channel score. \\
$U_{\mathrm{broker}}$ & scalar & Broker-channel score. \\
\end{localvartable}

\begin{breakablealgorithm}
\caption{\procChooseChannel}
\label{alg:choosechannel}
\begin{algorithmic}[1]
\Require Agent $i$ state $(s_i,\chi_i)$, broker reputation $\mathrm{rep}_b$, RNG.
\State $U_{\mathrm{self}}\gets s_{i,\mathrm{self}}$.
\State $U_{\mathrm{broker}}\gets \chi_i s_{i,\mathrm{broker}}+(1-\chi_i)\mathrm{rep}_b$.
\If{$U_{\mathrm{self}}>U_{\mathrm{broker}}$}
    \State \Return $\mathrm{self}$.
\EndIf
\If{$U_{\mathrm{broker}}>U_{\mathrm{self}}$}
    \State \Return $\mathrm{broker}$.
\EndIf
\State \Return $\mathrm{self}$ w.p. $1/2$, else $\mathrm{broker}$ \Comment{$U_{\mathrm{self}}=U_{\mathrm{broker}}$}.
\end{algorithmic}
\end{breakablealgorithm}

\section{One Period}

$p_{\mathrm{roster}}$ & scalar & Roster-churn probability. \\

\begin{breakablealgorithm}
\caption{\procPeriodUpdate}
\label{alg:period}
\begin{algorithmic}[1]
\Require Previous state $S^{t-1}$, period $t$, parameters $\vartheta$.
\Ensure Updated state $S^t$ and accepted records $\mathcal R^t$.
\State Clear accumulators, set $M_i^t\gets\emptyset$ for all $i$, and set $D^t\gets\emptyset$.
\State Refresh roster: remove each roster member with probability
$p_{\mathrm{roster}}$, then replenish uniformly to size $\lceil\alpha_R N\rceil$.
\State $\EG^t\gets
(\EG^t\setminus\{\{i,N+1\}:i\in[N]\})
\cup\{\{i,N+1\}:i\in\mathrm{Roster}^t\}$.
\State $\mathcal D^t\gets\emptyset$.
\For{$i=1,\ldots,N$}
    \State Draw $d_i^t\sim\mathrm{Binomial}(K,p_{\mathrm{demand}})$.
    \If{$d_i^t>0$}
        \State $z_i^t\gets\Call{\procChooseChannel}{i,\mathrm{rep}_b^{t-1}}$.
        \State Append $(i,z_i^t,d_i^t)$ to $\mathcal D^t$.
        \If{$z_i^t=\mathrm{broker}$}
            \State Add $i$ to $D^t$.
        \EndIf
    \EndIf
\EndFor
\State $\EG^t\gets
(\EG^t\setminus\{\{i,N+1\}:i\in[N]\})
\cup\{\{i,N+1\}:i\in\mathrm{Roster}^t\cup D^t\}$.
\State $\Call{\procTrainPredictors}{S,t,\vartheta}$.
\State $\mathcal R^t\gets\Call{\procOfferMarket}{S,\mathcal D^t,\vartheta}$.
\State Synchronize broker-node edges to $\mathrm{Roster}^t$, $D^t$, and current broker-channel matches.
\State $\Call{\procUpdateSatisfaction}{S,\mathcal D^t,\mathcal R^t}$.
\If{$D^t\neq\emptyset$}
    \State $\mathrm{rep}_b^t\gets |D^t|^{-1}\sum_{i\in D^t}s_{i,\mathrm{broker}}^t$.
\Else
    \State $\mathrm{rep}_b^t\gets \mathrm{rep}_b^{t-1}$.
\EndIf
\If{$t\bmod \Delta_{\mathrm{net}}=0$}
    \State \diag{Recompute broker betweenness, Burt constraint, and effective size on the post-matching, pre-turnover graph $G^t$.}
\EndIf
\State \diag{$Y^t\gets\Call{\procCollectMeasures}{S^t,\mathcal R^t,\vartheta}$ on the post-matching, pre-turnover state.}
\For{$i=1,\ldots,N$}
    \If{agent $i$ exits with probability $\eta_{\mathrm{exit}}$}
        \State $\Call{\procInitializeAgent}{S^t,i,t,\vartheta}$.
    \EndIf
\EndFor
\State Synchronize broker-node edges to the post-turnover roster, clients, and active matches.
\State \Return $(S^t,\mathcal R^t,Y^t)$.
\end{algorithmic}
\end{breakablealgorithm}

\section{Offer Market}

\begin{localvartable}{\procOfferMarket}
$\hat q_i^t$ & $\R^d\to\R$ & Agent predictor. \\
$\hat q_b^t$ & $\R^{d_b}\to\R$ & Broker predictor. \\
$\mathcal N_G^t(i)$ & subset of $[N]$ & Non-broker graph neighbors of $i$. \\
$U_i^t$ & subset of $[N]$ & Self-search stranger pool independently sampled for demander $i$. \\
$n_{\mathrm{strangers}}$ & integer & Per-demander stranger-pool size. \\
$v_{i\to j}^{\mathrm{self},t}$ & scalar & Self-offer score. \\
$\Omega_i$ & integer & Residual offer quota for demander $i$. \\
$\Omega$ & $\mathbb Z_{\ge0}^N$ & Residual quota vector. \\
$O^t$ & map & Directed offer book. \\
$O_{ij}^t$ & offer record & Record $(i,j,z,v)$ for offer $i\to j$. \\
$\mathcal P_O^t$ & set & Unordered pairs represented in $O^t$. \\
$\mathcal U_i^{\mathrm{self}}$ & subset of $[N]$ & Self-search candidate set. \\
$\mathcal I_i^{\mathrm{self}}$ & list & Self-search candidates sorted by score. \\
$\mathcal P_b^t$ & set & Feasible unordered broker-pair set. \\
$\mathcal I_{b,i}$ & list & Accessible broker candidates for client $i$, sorted by $v_b^t$. \\
$\mathcal B_b^t$ & list & Selected directed broker offers. \\
$o,o',o_{ij},o_{ji}$ & offer records & Offer records; fields include $o.z$ and $o.\mathrm{from}$. \\
$z^\star$ & channel & Accepted-record channel. \\
$\alpha,\beta$ & indices & Ordered endpoints induced by one unordered broker pair. \\
$\iota$ & $\{0,1\}$ & Return value of \procAddOffer. \\
\end{localvartable}

The symmetric broker score is
\[
v_b^t(i,j)=\hat q_b^t\!\left(\psi_b(x_i^t,x_j^t)\right).
\]
Let $\mathcal N_G^t(i)=\{j\in[N]:(i,j)\in \EG^t\}$ exclude the broker node.
The self-search proposal score is defined only for observed graph neighbors and
the demander-specific stranger pool:
\[
v_{i\to j}^{\mathrm{self},t}=
\begin{cases}
\bar q_{ij}^t, & j\in\mathcal N_G^t(i),\ C_{ij}^t>0,\\
\hat q_i^t(x_j^t), & j\in U_i^t\setminus\mathcal N_G^t(i),\\
\bot, & \text{otherwise}.
\end{cases}
\]
Thus direct pair means dominate predictor extrapolations when pair-specific
experience exists; $\bot$ denotes no valid self-search score. The receiver-side
acceptance score for a one-sided offer is
\[
w_{j\leftarrow i}^t=
\begin{cases}
\bar q_{ji}^t, & (i,j)\in \EG^t,\ C_{ji}^t>0,\\
\hat q_j^t(x_i^t), & \text{otherwise}.
\end{cases}
\]

The market creates at most $d_i^t$ outgoing offers from demander $i$.
Receivers have no per-period capacity constraint; acceptance is pairwise, so a
receiver may accept multiple offers in the same period. A demanded position can
remain unmatched if no acceptable offer is formed.

\begin{breakablealgorithm}
\caption{\procAddOffer}
\label{alg:addoffer}
\begin{algorithmic}[1]
\Require Offer book $O^t$, unordered-pair set $\mathcal P_O^t$, offer $(i,j,z,v)$.
\If{$O_{ij}^t$ is defined}
    \State \Return $0$.
\EndIf
\State $O_{ij}^t\gets(i,j,z,v)$.
\If{$O_{ji}^t$ is undefined}
    \State $\mathcal P_O^t\gets\mathcal P_O^t\cup\{\{i,j\}\}$.
\EndIf
\State \Return $1$.
\end{algorithmic}
\end{breakablealgorithm}

\begin{breakablealgorithm}
\caption{\procOfferMarket}
\label{alg:market}
\begin{algorithmic}[1]
\Require State $S$, demand records $\mathcal D^t$, parameters $\vartheta$.
\Ensure Accepted records $\mathcal R^t$.
\State For all $(i,z_i,d_i)\in\mathcal D^t$, set residual offer quota $\Omega_i\gets d_i$.
\State $O^t\gets\emptyset$, $\mathcal P_O^t\gets\emptyset$, $\mathcal R^t\gets\emptyset$.
\For{each $(i,\mathrm{self},d_i)\in\mathcal D^t$ with $\Omega_i>0$}
    \State Draw $U_i^t\subset[N]$ uniformly without replacement,
    $|U_i^t|=\min(n_{\mathrm{strangers}},N)$.
    \State $\mathcal U_i^{\mathrm{self}}\gets
    \{j\in\mathcal N_G^t(i):C_{ij}^t>0\}
    \cup (U_i^t\setminus\mathcal N_G^t(i))$.
    \State Remove from $\mathcal U_i^{\mathrm{self}}$ the elements of $\{i\}\cup M_i^t$.
    \State $\mathcal I_i^{\mathrm{self}}\gets\argsort_{j\in\mathcal U_i^{\mathrm{self}}:\ v_{i\to j}^{\mathrm{self},t}>r}
    (-v_{i\to j}^{\mathrm{self},t})$.
    \For{the first $\min(\Omega_i,|\mathcal I_i^{\mathrm{self}}|)$ agents $j$ in $\mathcal I_i^{\mathrm{self}}$}
        \State $\Call{\procAddOffer}{O^t,\mathcal P_O^t,i,j,\mathrm{self},v_{i\to j}^{\mathrm{self},t}}$.
    \EndFor
\EndFor
\State $\Ab^t\gets\mathrm{Roster}^t\cup D^t$.
\State $D_+^t\gets\{i:(i,\mathrm{broker},d_i)\in\mathcal D^t,\ \Omega_i>0\}$.
\State $\mathcal P_b^t\gets\{\{i,j\}:i\ne j,\ (i\in D_+^t,j\in\Ab^t)\ \text{or}\ (j\in D_+^t,i\in\Ab^t)\}$.
\State $\mathcal B_b^t\gets\emptyset$.
\For{each $i\in D_+^t$}
    \State $\mathcal I_{b,i}\gets\argsort_{j\in\Ab^t:\{i,j\}\in\mathcal P_b^t,\ v_b^t(i,j)>r}(-v_b^t(i,j))$.
    \For{the first $\min(\Omega_i,|\mathcal I_{b,i}|)$ agents $j$ in $\mathcal I_{b,i}$}
        \State Append $(i,j,v_b^t(i,j))$ to $\mathcal B_b^t$.
    \EndFor
\EndFor
\State Sort $\mathcal B_b^t$ by descending $v_b^t$ with the deterministic unordered-pair tie key.
\For{each $(i,j,v)\in\mathcal B_b^t$}
    \If{$\Omega_i>0$}
        \State $\iota\gets\Call{\procAddOffer}{O^t,\mathcal P_O^t,i,j,\mathrm{broker},v}$.
        \State $\Omega_i\gets\Omega_i-\iota$.
    \EndIf
\EndFor
\For{each $\{i,j\}\in\mathcal P_O^t$}
    \State $o_{ij}\gets O_{ij}^t$ if defined, else $\emptyset$; $o_{ji}\gets O_{ji}^t$ if defined, else $\emptyset$.
    \If{$o_{ij}\ne\emptyset$ and $o_{ji}\ne\emptyset$}
        \State $\Call{\procFinalizeMatch}{S,\mathcal R^t,o_{ij},o_{ji}}$.
    \ElsIf{$o_{ij}\ne\emptyset$ and $w_{j\leftarrow i}^t>r$}
        \State $\Call{\procFinalizeMatch}{S,\mathcal R^t,o_{ij},\emptyset}$.
    \ElsIf{$o_{ji}\ne\emptyset$ and $w_{i\leftarrow j}^t>r$}
        \State $\Call{\procFinalizeMatch}{S,\mathcal R^t,o_{ji},\emptyset}$.
    \EndIf
\EndFor
\State \Return $\mathcal R^t$.
\end{algorithmic}
\end{breakablealgorithm}

\begin{breakablealgorithm}
\caption{\procFinalizeMatch}
\label{alg:finalizematch}
\begin{algorithmic}[1]
\Require State $S$, accepted-record list $\mathcal R^t$, offer
$o=(i,j,z,v)$, optional reverse offer $o'$.
\If{$j\in M_i^t$}
    \State \Return.
\EndIf
\State Draw $q_{ij}^t=Q+f(x_i^t,x_j^t)+\varepsilon$, $\varepsilon\sim \mathcal N(0,\sigma_\varepsilon^2)$.
\State Append columns $x_j^t$ to $\Xi_i$ and $x_i^t$ to $\Xi_j$; append
$q_{ij}^t$ to $y_i$ and $y_j$; increment $n_i$ and $n_j$.
\State Update $\bar q_{ij},C_{ij}$ and $\bar q_{ji},C_{ji}$.
\State Add edge $(i,j)$ to $G^t$; add $j$ to $M_i^t$ and $i$ to $M_j^t$.
\If{$z=\mathrm{broker}$ or $(o'\ne\emptyset$ and $o'.z=\mathrm{broker})$}
    \State Append column $[x_i^t;x_j^t]$ to $\Xi_b$; append $q_{ij}^t$ to $y_b$; increment $n_b$.
    \State $z^\star\gets\mathrm{broker}$.
\Else
    \State $z^\star\gets\mathrm{self}$.
\EndIf
\State Append accepted record $(i,j,z^\star,q_{ij}^t,v,o,o')$ to $\mathcal R^t$.
\end{algorithmic}
\end{breakablealgorithm}

\section{Satisfaction and Measures}

\begin{localvartable}{satisfaction}
$Y_i$ & scalar & Agent $i$ period satisfaction numerator. \\
$n_{i,b}$ & integer & Accepted broker offers credited to demander $i$. \\
$\tilde q_i^t$ & scalar & Channel-specific satisfaction input. \\
$\zeta$ & accepted record & Iterator over $\mathcal R^t$. \\
$\mathbf{1}\{\cdot\}$ & indicator & Equals $1$ when its condition is true and $0$ otherwise. \\
\end{localvartable}

\begin{breakablealgorithm}
\caption{\procUpdateSatisfaction}
\label{alg:satisfaction}
\begin{algorithmic}[1]
\Require State $S$, demand records $\mathcal D^t$, accepted records $\mathcal R^t$.
\For{each $(i,z_i,d_i)\in\mathcal D^t$}
    \State $Y_i\gets \sum_{\zeta\in\mathcal R^t}\sum_{o\in\{\zeta.o_1,\zeta.o_2\}}
    \mathbf{1}\{o\ne\emptyset,\ o.\mathrm{from}=i\}\,\zeta.q$.
    \State $n_{i,b}\gets \sum_{\zeta\in\mathcal R^t}\sum_{o\in\{\zeta.o_1,\zeta.o_2\}}
    \mathbf{1}\{o\ne\emptyset,\ o.\mathrm{from}=i,\ o.z=\mathrm{broker}\}$.
    \If{$z_i=\mathrm{self}$}
        \State $\tilde q_i^t\gets Y_i/d_i-c_s$.
    \Else
        \State $\tilde q_i^t\gets (Y_i-\phi n_{i,b})/d_i$; $\chi_i^t\gets1$.
    \EndIf
    \State $s_{i,z_i}^t\gets(1-\omega)s_{i,z_i}^{t-1}+\omega\tilde q_i^t$.
\EndFor
\end{algorithmic}
\end{breakablealgorithm}

\paragraph{\diag{\procCollectMeasures.}}
\diag{$\Call{\procCollectMeasures}{S^t,\mathcal R^t,\vartheta}$ is a pure measure-collection
function called after matching and before entry/exit turnover. It records demand, outsourcing, accepted matches, realized output,
selected-sample prediction quality, deterministic holdout prediction quality,
broker access and assessment counts, satisfaction, reputation, roster size,
broker access size, agent-node degree summaries excluding the broker node,
and cached broker network measures. The measures themselves do not feed
back into $S^{t+1}$.}
