Offer Making And Acceptance

Inputs already fixed for the period:
    D_self                 # agents with active self-search demand
    D_broker               # agents with active broker demand
    d[i]                   # active demand count for each demander, 1 <= d[i] <= K
    r                      # reservation threshold
    n_strangers = 10       # fixed period-level stranger-pool size
    broker.roster
    broker.current_clients
    agents 1..N

Definitions:
    offer_book[{i,j}] stores directed offers for unordered pair {i,j}
    broker_value({i,j}) = qhat_b({i,j})
        # symmetric by specification: broker_value({i,j}) = broker_value({j,i})

    self_value(i,j):
        if i has a stored partner mean for j:
            return partner_mean(i,j)
        else:
            return agent_prediction(i,j)

    receiver_value(j,i):
        if j has a stored partner mean for i:
            return partner_mean(j,i)
        else:
            return agent_prediction(j,i)

    add_offer!(from=i, to=j, channel=c, predicted_value=v):
        require i != j
        key = unordered_pair(i,j)
        if offer_book[key] already contains an offer from i to j:
            return false
        offer_book[key].add((from=i, to=j, channel=c, predicted_value=v))
        return true


Step 1. Create the period-level stranger pool

    S = sample_without_replacement({1, ..., N}, min(n_strangers, N))


Step 2. Self-search demanders make offers from self-search lists

    for each i in D_self:

        sent = 0

        known_candidates =
            known_self_search_counterparties(i)

        stranger_candidates =
            {j in S : j not in known_candidates}

        candidates =
            known_candidates union stranger_candidates

        ranked_candidates =
            sort candidates descending by self_value(i,j)

        for each j in ranked_candidates:

            if sent == d[i]:
                break

            v = self_value(i,j)

            if v <= r:
                continue

            if add_offer!(from=i, to=j, channel=self, predicted_value=v):
                sent += 1


Step 3. Broker-demanders make offers from broker lists

    remaining[i] = d[i] for each i in D_broker

    broker_access =
        {j in 1..N : j in broker.roster
                    or j in broker.current_clients}

    broker_pairs =
        unordered pairs {i,j} such that:
            i != j
            and (
                (i in D_broker and j in broker_access)
                or
                (j in D_broker and i in broker_access)
            )

    ranked_broker_pairs =
        sort broker_pairs descending by broker_value({i,j})

    for each unordered pair {i,j} in ranked_broker_pairs:

        v = broker_value({i,j})

        if v <= r:
            continue

        if i in D_broker and remaining[i] > 0 and j in broker_access:
            if add_offer!(from=i, to=j, channel=broker, predicted_value=v):
                remaining[i] -= 1

        if j in D_broker and remaining[j] > 0 and i in broker_access:
            if add_offer!(from=j, to=i, channel=broker, predicted_value=v):
                remaining[j] -= 1


Step 4. Accept or reject offers

    accepted_pairs = []

    for each unordered pair {i,j} with offer_pair in offer_book:

        if offer_pair contains both i -> j and j -> i:

            # Reciprocal offers are accepted by construction.
            accepted_pairs.append(({i,j}, offer_pair, reciprocal=true))
            continue

        offer = the single directed offer i -> j

        if receiver_value(j,i) > r:
            accepted_pairs.append(({i,j}, offer_pair, reciprocal=false))


Output:
    accepted_pairs
